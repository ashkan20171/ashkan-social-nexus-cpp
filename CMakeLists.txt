cmake_minimum_required(VERSION 3.16)
project(AshkanSocialNexus LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(SQLite3 QUIET)
if(NOT SQLite3_FOUND)
 message(FATAL_ERROR "SQLite3 not found. Set VCPKG_ROOT and use the windows-vcpkg preset, or set SQLite3_INCLUDE_DIR and SQLite3_LIBRARY. See README.md")
endif()
find_package(OpenSSL REQUIRED)
add_executable(AshkanSocialNexus src/main.cpp)
target_link_libraries(AshkanSocialNexus PRIVATE SQLite::SQLite3 OpenSSL::Crypto)
target_compile_definitions(AshkanSocialNexus PRIVATE NEXUS_WEB_DIR="${CMAKE_SOURCE_DIR}/web")
if(WIN32)
 target_link_libraries(AshkanSocialNexus PRIVATE ws2_32)
endif()
add_custom_command(TARGET AshkanSocialNexus POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/web $<TARGET_FILE_DIR:AshkanSocialNexus>/web)

include(CTest)
if(BUILD_TESTING)
 add_executable(nexus_security_tests tests/security_tests.cpp)
 target_include_directories(nexus_security_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
 add_test(NAME security_validation COMMAND nexus_security_tests)
 add_executable(nexus_http_tests tests/http_tests.cpp)
 target_include_directories(nexus_http_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
 add_test(NAME http_validation COMMAND nexus_http_tests)
endif()

if(BUILD_TESTING)
 add_executable(nexus_api_contract_tests tests/api_contract_tests.cpp)
 target_include_directories(nexus_api_contract_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
 add_test(NAME api_contract_validation COMMAND nexus_api_contract_tests)
endif()

if(BUILD_TESTING)
 add_executable(nexus_database_tests tests/database_tests.cpp)
 target_include_directories(nexus_database_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
 target_link_libraries(nexus_database_tests PRIVATE SQLite::SQLite3)
 add_test(NAME database_migrations COMMAND nexus_database_tests)
endif()

if(BUILD_TESTING)
 add_executable(nexus_audit_tests tests/audit_tests.cpp)
 target_include_directories(nexus_audit_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
 target_link_libraries(nexus_audit_tests PRIVATE SQLite::SQLite3)
 add_test(NAME audit_triggers COMMAND nexus_audit_tests)
endif()
